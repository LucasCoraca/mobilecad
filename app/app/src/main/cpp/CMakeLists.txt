cmake_minimum_required(VERSION 3.22)
project(fcbridge CXX)
set(CMAKE_CXX_STANDARD 17)
# CPython for Android, built by engine/build_python.sh into $FCDEPS/<abi>/python/prefix (FCDEPS as in engine/env.sh)
if(DEFINED ENV{FCDEPS})
  set(FCDEPS $ENV{FCDEPS})
else()
  set(FCDEPS $ENV{HOME}/Android/fcdeps)
endif()
set(PY ${FCDEPS}/${ANDROID_ABI}/python/prefix)
# Without a native build tree (only the staged runtime): compile against python.org's CPython 3.14.4 Android package
# (same version and soname as ours; only its headers and libpython link stub are used, the APK ships our libpython),
# unpacked to ~/Android/pyprefix/<abi>/prefix
if(NOT EXISTS ${PY}/include/python3.14/Python.h AND EXISTS $ENV{HOME}/Android/pyprefix/${ANDROID_ABI}/prefix/include/python3.14/Python.h)
  set(PY $ENV{HOME}/Android/pyprefix/${ANDROID_ABI}/prefix)
endif()
add_library(fcbridge SHARED fcbridge.cpp)
target_include_directories(fcbridge PRIVATE ${PY}/include/python3.14)
target_link_libraries(fcbridge ${PY}/lib/libpython3.14.so log dl)
# mcxr: XR mode (MainActivity starts XrActivity on a headset). The XR core in xr/core (shared with the desktop test
# bench in xr/desktop) + its Android glue; the OpenXR loader comes from the Khronos AAR (prefab)
find_package(OpenXR REQUIRED CONFIG)
set(XRCORE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../xr/core)
add_library(mcxr SHARED xr_android.cpp ${XRCORE}/scene.cpp ${XRCORE}/renderer.cpp ${XRCORE}/interaction.cpp ${XRCORE}/hands.cpp ${XRCORE}/gltf_model.cpp ${XRCORE}/xr_session.cpp)
target_include_directories(mcxr PRIVATE ${XRCORE})
target_link_libraries(mcxr OpenXR::openxr_loader EGL GLESv3 android log)
# mcmesh: display meshes of shapes, natively (see mcmesh.cpp).  Links OCCT from the staged FreeCAD runtime and compiles
# against OCCT's headers (engine/occt_headers.sh unpacks them from the source archive); without them the bridge
# meshes in Python as before.
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(FCRT ${CMAKE_CURRENT_SOURCE_DIR}/../../arm64/assets/freecad-runtime/lib)
else()
  set(FCRT ${CMAKE_CURRENT_SOURCE_DIR}/../../x86/assets/freecad-runtime/lib)
endif()
set(OCCT_INC $ENV{HOME}/Android/occt-inc)
if(EXISTS ${OCCT_INC}/Standard.hxx AND EXISTS ${FCRT}/libTKMesh.so)
  add_library(mcmesh SHARED mcmesh.cpp)
  target_include_directories(mcmesh SYSTEM PRIVATE ${OCCT_INC})
  target_compile_options(mcmesh PRIVATE -O2 -Wno-deprecated-declarations)
  foreach(tk TKMesh TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
    target_link_libraries(mcmesh ${FCRT}/lib${tk}.so)
  endforeach()
else()
  message(WARNING "mcmesh not built: OCCT headers (${OCCT_INC}) or runtime (${FCRT}) missing")
endif()
